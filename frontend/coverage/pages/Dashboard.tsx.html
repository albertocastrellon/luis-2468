/** El nombre registrado se muestra en estilo título (Ana López → Ana López). */
function formatName(fullName?: string): string {
  return (
    fullName
      ?.trim()
      .split(" ")
      .filter(Boolean)
      .map((word) => word.charAt(0).toUpperCase() + word.slice(1).toLowerCase())
      .join(" ") ?? ""
  );
}
